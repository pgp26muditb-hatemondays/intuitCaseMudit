from intuit_ies import Agent, WorkforceAPI, ProjectAPI
from intuit_ies.ai import ComplianceEngine

class SafetyComplianceAgent(Agent):
  def __init__(self):
    self.workforce = WorkforceAPI()
    self.projects = ProjectAPI()
    self.engine = ComplianceEngine("OSHA")

  async def monitor(self):
    sites = await self.projects.active_sites()
    for site in sites:
      crew = await self.workforce.get_crew(site.id)
      violations = self.engine.check(crew, site)
      if violations:
        await self.alert(site.pm, violations)
        await self.auto_file(violations)
